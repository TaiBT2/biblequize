🔧 Backend changes (Sprint 6)
-- V53__group_activity_feed.sql
CREATE TABLE group_activity (
  id BIGINT PK,
  group_id BIGINT FK,
  actor_user_id BIGINT FK,
  type ENUM(
    'MASTERY_COMPLETED',
    'QUIZ_SET_CREATED',
    'QUIZ_SET_UPDATED',
    'LIVE_ROOM_STARTED',
    'LIVE_ROOM_ENDED',
    'ANNOUNCEMENT_POSTED',
    'MEMBER_JOINED',
    'MEMBER_LEFT',
    'SCHEDULED_QUIZ_CREATED'
  ),
  metadata JSON,
  created_at TIMESTAMP,
  INDEX (group_id, created_at DESC)
);

-- V54__group_pulse.sql
CREATE TABLE group_pulse_snapshot (
  group_id BIGINT PK,
  pulse_score DECIMAL(3,2),  -- 0.00-1.00
  active_ratio DECIMAL(3,2),
  live_rooms_count INT,
  new_content_count INT,
  status ENUM('STRONG','MEDIUM','WEAK'),
  computed_at TIMESTAMP
);

-- Cron: GroupPulseScheduler
-- @Scheduled(cron = "0 0 1 * * *")  // 1am daily
public void computeAllGroupPulses() {
  for (each group) {
    pulse = activeRatio*0.5 + liveRooms*0.3 
          + newContent*0.2;
    save snapshot;
  }
}
